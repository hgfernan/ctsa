cmake_minimum_required(VERSION 3.10)
project(MyTests)

# Common dependencies
set(MY_LIBRARY your_library_name)
set(INCLUDE_DIR path/to/includes)

# Collect all test source files
file(GLOB TEST_SOURCES "path/to/tests/*.cpp")

# Function to add a test executable
function(add_test_executable test_source)
    get_filename_component(test_name ${test_source} NAME_WE)
    add_executable(${test_name} ${test_source})
    target_link_libraries(${test_name} PRIVATE ${MY_LIBRARY})
    target_include_directories(${test_name} PRIVATE ${INCLUDE_DIR})
endfunction()

# Generate executables for all tests
foreach(test_source ${TEST_SOURCES})
    add_test_executable(${test_source})
endforeach()
